cmake_minimum_required(VERSION 3.22)
project(task3_agent_standalone LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
find_package(fastdds 3 REQUIRED)
find_package(fastcdr REQUIRED)
add_executable(task3_agent ../agent_main.cpp ../Collectors.cpp ../RuntimeIdentity.cpp ../DdsTransport.cpp ../Protocol.cpp)
target_include_directories(task3_agent PRIVATE ..)
target_link_libraries(task3_agent PRIVATE Qt6::Core Qt6::Network fastdds fastcdr)
if(WIN32)
  target_link_libraries(task3_agent PRIVATE iphlpapi psapi advapi32)
  add_custom_command(TARGET task3_agent POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/windows-time-status.ps1 $<TARGET_FILE_DIR:task3_agent>)
endif()
